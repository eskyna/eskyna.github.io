{{- /* Responsive Image Partial mit Hugo Image Processing Parameter: - src: Bild-Pfad (relativ von /static oder /content) - alt: Alt-Text (erforderlich) - title: Title-Attribut (optional) - sizes: CSS Media Query Sizes (optional, default: responsive) - loading: lazy | eager (optional) - class: CSS-Klasse (optional) - decoding: async | auto (optional) - fetchpriority: high | low | auto (optional) - maxWidth: maximale Breite in px (optional) */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $title := .title | default $alt -}} {{- $class := .class -}} {{- $loading := .loading | default "lazy" -}} {{- $decoding := .decoding -}} {{- $fetchpriority := .fetchpriority -}} {{- $sizes := .sizes | default "(max-width: 640px) 100vw, (max-width: 1024px) 80vw, 1200px" -}} {{- $width := .width -}} {{- $height := .height -}} {{- $maxWidth := .maxWidth | default 1200 -}} {{- /* Externe URLs nicht verarbeiten */ -}} {{- $isAbsolute := or (hasPrefix $src "http://") (hasPrefix $src "https://") (hasPrefix $src "data:") -}} {{- if $isAbsolute -}} {{- /* Externe Bilder: einfach anzeigen */ -}} {{- else -}} {{- /* Lokale Bilder */ -}} {{- $ext := lower (path.Ext $src) -}} {{- $isPNG := or (eq $ext ".png") (eq $ext ".jpg") (eq $ext ".jpeg") -}} {{- if $isPNG -}} {{- /* Versuche als Page Resource zu laden (für Content-Bilder) */ -}} {{- $img := .Page.Resources.GetMatch $src -}} {{- if $img -}} {{- /* Generiere responsive Größen via Hugo Image Processing */ -}} {{- $imgSizes := slice 320 640 960 1200 1600 2000 -}} {{- $srcset := slice -}} {{- $srcsetWebp := slice -}} {{- $srcsetAvif := slice -}} {{- $imgWidth := $img.Width -}} {{- $imgHeight := $img.Height -}} {{- range $imgSizes -}} {{- if le . $imgWidth -}} {{- $scaledHeight := math.Round (div (mul . $imgHeight) $imgWidth) -}} {{- $resizeDim := printf "%dx%d" . (int $scaledHeight) -}} {{- $png := $img.Resize $resizeDim -}} {{- $srcset = $srcset | append (printf "%s %dw" $png.RelPermalink .) -}} {{- $webp := $img.Resize (printf "%s webp" $resizeDim) -}} {{- $srcsetWebp = $srcsetWebp | append (printf "%s %dw" $webp.RelPermalink .) -}} {{- $avif := $img.Resize (printf "%s avif" $resizeDim) -}} {{- $srcsetAvif = $srcsetAvif | append (printf "%s %dw" $avif.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $pngFallback := $img.Resize (printf "%dx auto" $maxWidth) -}} {{- else -}} {{- /* Statisches Bild: nutze vorkonvertierte WebP/AVIF aus convert-images-build */ -}} {{- $stem := strings.TrimSuffix $ext $src -}} {{- $webpSrc := printf "%s.webp" $stem -}} {{- $avifSrc := printf "%s.avif" $stem -}} {{- $hasWebp := fileExists (printf "static/%s" $webpSrc) -}} {{- $hasAvif := fileExists (printf "static/%s" $avifSrc) -}} {{- /* Responsive srcset für Bilder mit generierten Größenvarianten */ -}} {{- $responsiveWidths := slice 640 960 1200 1600 -}} {{- $hasResponsive := false -}} {{- /* Prüfe ob mindestens eine Responsive-Variante existiert */ -}} {{- $testPath := printf "%s-%d.webp" $stem (index $responsiveWidths 0) -}} {{- if (fileExists (printf "static/%s" $testPath)) -}} {{- $hasResponsive = true -}} {{- end -}} {{- if $hasResponsive -}} {{- $avifSrcset := slice -}} {{- $webpSrcset := slice -}} {{- range $responsiveWidths -}} {{- $wAvif := printf "%s-%d.avif" $stem . -}} {{- $wWebp := printf "%s-%d.webp" $stem . -}} {{- if (fileExists (printf "static/%s" $wAvif)) -}} {{- $avifSrcset = $avifSrcset | append (printf "%s %dw" ($wAvif | relURL) .) -}} {{- end -}} {{- if (fileExists (printf "static/%s" $wWebp)) -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" ($wWebp | relURL) .) -}} {{- end -}} {{- end -}} {{- if gt (len $avifSrcset) 0 -}} {{- end -}} {{- if gt (len $webpSrcset) 0 -}} {{- end -}} {{- else -}} {{- if or $hasAvif $hasWebp -}} {{- if $hasAvif -}}{{- end -}} {{- if $hasWebp -}}{{- end -}} {{- else -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Andere Formate (webp, avif, gif, svg): einfach anzeigen */ -}} {{- end -}} {{- end -}}
